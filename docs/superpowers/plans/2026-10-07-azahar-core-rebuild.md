# Azahar 3DS Core Rebuild Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the hand-forked `Cores/Citra` 3DS core with `Cores/Azahar`, a core that builds upstream azahar-emu through CMake into an xcframework and adds only Apple glue, shipping alongside PVEmuThree with a data migrator.

**Architecture:** A `Provenance-Emu/azahar` fork branch `provenance` carries a small upstreamable patch stack over azahar-emu master. `build_azahar_core.py` compiles the emulator libraries for iOS/tvOS device and simulator with CMake+Ninja and merges them into `PVlibAzahar.xcframework`. `PVAzahar.xcodeproj` (generated by xcodegen from `project.yml`) compiles ~15 glue/bridge files against that xcframework and links the PV* packages. Rendering goes through a `CAMetalLayer`-backed `Frontend::EmuWindow` and the app's embedded dynamic `MoltenVK.framework`.

**Tech Stack:** C++20/Objective-C++ (azahar API, PVCoreObjCBridge), Swift 6 (PVEmulatorCore subclass, options, migrator), CMake 3.25+/Ninja, xcodegen, swiftgen, Python 3 build script, XCTest.

**Spec:** `docs/superpowers/specs/2026-10-07-azahar-core-rebuild-design.md`

## Global Constraints

- Minimum deployment: iOS 17.0, tvOS 17.0. Every file must compile for both platforms; use `#if os(tvOS)` / `#if !TARGET_OS_TV` guards.
- Core identifier `com.provenance.core.azahar`; principle class `PVAzahar.PVAzaharCore`; system `com.provenance.3ds`; `PVJITRequirement` = `optional`; license `GPL-2.0-or-later`.
- User data root is `<app Documents>/Azahar/` (iOS and tvOS). emuThree legacy root is `<app Documents>/` on iOS and `<Library/Caches>/` on tvOS.
- No shadowed emulator sources: every change to azahar code is a commit on the fork's `provenance` branch, recorded in `Cores/Azahar/PATCHES.md`.
- System identifiers are compared through `SystemIdentifier` (Swift) or a single `static NSString * const` (ObjC), never inline string literals.
- Notification names, if any, follow the `FOUNDATION_EXPORT NSNotificationName const` rule in CLAUDE.md.
- `Scripts/audits/check_pbxproj_sources.py` must pass: every new source file beside compiled ones is listed in the project (xcodegen handles this when `project.yml` globs the directory; regenerate after adding files).
- Commits use conventional prefixes and end with `Co-Authored-By: Claude Fable 5.1 <noreply@anthropic.com>`.
- Subagents work on their own branch/worktree and never `git reset`, `rebase`, `push` or touch `develop` in the Provenance repo. Pushing the `provenance` branch of the `Provenance-Emu/azahar` fork is the one allowed push, in Task 1 only.
- `Cores/Azahar/build/` is gitignored.

- [ ] **Step 2: Write `build_azahar_core.py`**

Model the file on `Cores/Dolphin/build_dolphin_core.py` (same `CONFIG` dict, `PLATFORMS`, argparse, `run()` helper, log colouring). The azahar-specific parts:

```python
#!/usr/bin/env python3
"""
Build the azahar 3DS emulator libraries for iOS/tvOS and package them as
PVlibAzahar.xcframework for Cores/Azahar/PVAzahar.xcodeproj.

Source: Cores/Azahar/azahar (Provenance-Emu/azahar, branch `provenance`; see PATCHES.md)
Output: Cores/Azahar/build/xcframework/PVlibAzahar.xcframework

Usage:
  build_azahar_core.py -p OS64            # one slice (iphoneos)
  build_azahar_core.py --all-platforms    # all four slices + xcframework
  build_azahar_core.py --clean -p TVOS
"""
import argparse, os, shutil, subprocess, sys, multiprocessing
from pathlib import Path

ROOT = Path(__file__).resolve().parent
SRC = ROOT / "azahar"
BUILD = ROOT / "build"
TOOLCHAIN = ROOT / "cmake" / "ios.toolchain.cmake"
LIB_NAME = "PVlibAzahar"
MOLTENVK_XCFW = ROOT.parents[1] / "MoltenVK" / "MoltenVK" / "static" / "MoltenVK.xcframework"

PLATFORMS = {
    "OS64":           {"sdk": "iphoneos",          "slice": "ios",      "min": "17.0", "mvk": "ios-arm64"},
    "SIMULATORARM64": {"sdk": "iphonesimulator",   "slice": "ios-sim",  "min": "17.0", "mvk": "ios-arm64_x86_64-simulator"},
    "TVOS":           {"sdk": "appletvos",         "slice": "tvos",     "min": "17.0", "mvk": "tvos-arm64_arm64e"},
    "SIMULATOR_TVOS": {"sdk": "appletvsimulator",  "slice": "tvos-sim", "min": "17.0", "mvk": "tvos-arm64_x86_64-simulator"},
}

CMAKE_OPTIONS = [
    "-DCMAKE_BUILD_TYPE=Release",
    "-DENABLE_QT=OFF", "-DENABLE_SDL2=OFF", "-DENABLE_WEB_SERVICE=OFF",
    "-DENABLE_SCRIPTING=OFF", "-DENABLE_GDBSTUB=OFF", "-DENABLE_OPENAL=OFF",
    "-DENABLE_CUBEB=OFF", "-DENABLE_LIBUSB=OFF", "-DENABLE_FFMPEG=OFF",
    "-DENABLE_COREAUDIO=ON", "-DENABLE_SOFTWARE_RENDERER=OFF", "-DENABLE_VULKAN=ON",
    "-DENABLE_TESTS=OFF", "-DENABLE_ROOM=OFF", "-DENABLE_DISCORD_RPC=OFF",
    "-DENABLE_LTO=ON", "-DUSE_SYSTEM_MOLTENVK=ON",
    "-DCITRA_WARNINGS_AS_ERRORS=OFF", "-DENABLE_COMPATIBILITY_LIST_DOWNLOAD=OFF",
]

# CMake targets whose static archives make up PVlibAzahar. Externals are pulled in
# transitively by `ninja <targets>`; we then harvest every .a under the build dir.
TARGETS = ["citra_core", "citra_common", "video_core", "audio_core", "network", "input_common"]

def configure(platform: str, verbose: bool) -> Path:
    p = PLATFORMS[platform]
    out = BUILD / platform
    out.mkdir(parents=True, exist_ok=True)
    mvk = MOLTENVK_XCFW / p["mvk"] / "libMoltenVK.a"
    if not mvk.exists():
        sys.exit(f"error: MoltenVK static slice missing: {mvk}")
    cmd = ["cmake", str(SRC), "-GNinja",
           f"-DCMAKE_TOOLCHAIN_FILE={TOOLCHAIN}", f"-DPLATFORM={platform}",
           f"-DDEPLOYMENT_TARGET={p['min']}", "-DENABLE_BITCODE=OFF", "-DENABLE_ARC=OFF",
           "-DENABLE_VISIBILITY=OFF", f"-DMOLTENVK_LIBRARY={mvk}",
           *CMAKE_OPTIONS]
    if shutil.which("ccache"):
        cmd += ["-DCMAKE_C_COMPILER_LAUNCHER=ccache", "-DCMAKE_CXX_COMPILER_LAUNCHER=ccache"]
    run(cmd, cwd=out, verbose=verbose)
    return out

def build(out: Path, verbose: bool):
    run(["ninja", f"-j{multiprocessing.cpu_count()}", *TARGETS], cwd=out, verbose=verbose)

def harvest(out: Path) -> list[Path]:
    libs = sorted(set(out.rglob("*.a")))
    # Exclude anything we never link (tests, tools) and MoltenVK itself (loaded dynamically at runtime).
    return [l for l in libs if not any(s in l.name for s in ("MoltenVK", "catch2", "Catch2"))]

def merge(platform: str, out: Path, libs: list[Path]) -> Path:
    p = PLATFORMS[platform]
    fw = BUILD / "xcframework" / f"{LIB_NAME}-{p['slice']}.framework"
    if fw.exists():
        shutil.rmtree(fw)
    (fw / "Headers").mkdir(parents=True)
    run(["libtool", "-static", "-no_warning_for_no_symbols", "-o", str(fw / LIB_NAME), *map(str, libs)])
    copy_headers(fw / "Headers")
    write_framework_plist(fw, p["sdk"], p["min"])
    return fw

def copy_headers(dst: Path):
    def cp_tree(src: Path, sub: str, exts=(".h", ".hpp", ".inc", ".inl")):
        for f in src.rglob("*"):
            if f.suffix in exts:
                rel = f.relative_to(src)
                (dst / sub / rel).parent.mkdir(parents=True, exist_ok=True)
                shutil.copy2(f, dst / sub / rel)
    cp_tree(SRC / "src", "")                                  # core/, common/, video_core/, audio_core/, network/, input_common/
    cp_tree(SRC / "externals" / "boost" / "boost", "boost")
    cp_tree(SRC / "externals" / "fmt" / "include" / "fmt", "fmt")
```

```objc
- (void)startEmulation {
    [self applySettingsFromOptions];
    [self setupRenderView];          // +Video, main thread; creates _window

        ELOG(@"[PVAzahar] no render window; cannot start");
        return;
    }
    _running = true; _paused = false;
    std::string romPath([self.romPath fileSystemRepresentation]);
    _emuThread = std::thread([self, romPath] {
        pthread_setname_np("Azahar Emulation");
        auto& system = Core::System::GetInstance();
        AzaharInput::RegisterFactories();
        Frontend::RegisterDefaultApplets(system);
        Network::Init();
        system.ApplySettings();
        const auto result = system.Load(*_window, romPath);
        if (result != Core::System::ResultStatus::Success) {
            ELOG(@"[PVAzahar] Core::System::Load failed: %d", static_cast<int>(result));
            _running = false;

                [self stopEmulationWithMessage:[NSString stringWithFormat:@"Azahar could not load this title (%d)", static_cast<int>(result)]];

            return;

        while (_running) {

                std::unique_lock lock(_jobMutex);
                while (!_jobs.empty()) { auto job = std::move(_jobs.front()); _jobs.pop_front(); lock.unlock(); job(); lock.lock(); }

- (BOOL)runOnEmuThreadAndWait:(std::function<void()>)job timeout:(NSTimeInterval)seconds {
    if (!_running) { return NO; }
```

```objc
#import "PVAzaharCoreBridge.h"
#import "PVAzaharCoreBridge+Private.h"
#import <PVLoggingObjC/PVLoggingObjC.h>
#include <fstream>
#include <vector>
#include "core/core.h"

static NSString * const PVAzaharErrorDomain = @"org.provenance-emu.PVAzahar";
static const NSTimeInterval PVAzaharStateTimeout = 10.0;

@implementation PVAzaharCoreBridge (Saves)

- (BOOL)saveStateToFileAtPath:(NSString *)path error:(NSError **)error {
    __block std::vector<u8> buffer;
    const BOOL ran = [self runOnEmuThreadAndWait:[&buffer] { buffer = Core::System::GetInstance().SaveStateBuffer(); } timeout:PVAzaharStateTimeout];
    if (!ran || buffer.empty()) {
        if (error) *error = [NSError errorWithDomain:PVAzaharErrorDomain code:1 userInfo:@{NSLocalizedDescriptionKey: @"Azahar did not produce a save state"}];
        return NO;
    }
    NSData *data = [NSData dataWithBytes:buffer.data() length:buffer.size()];
    return [data writeToFile:path options:NSDataWritingAtomic error:error];
}

- (BOOL)loadStateFromFileAtPath:(NSString *)path error:(NSError **)error {
    NSData *data = [NSData dataWithContentsOfFile:path options:0 error:error];
    if (!data) { return NO; }
    std::vector<u8> buffer((const u8 *)data.bytes, (const u8 *)data.bytes + data.length);
    __block bool ok = false;
    const BOOL ran = [self runOnEmuThreadAndWait:[&] { ok = Core::System::GetInstance().LoadStateBuffer(std::move(buffer)); } timeout:PVAzaharStateTimeout];
    if (!ran || !ok) {
        if (error) *error = [NSError errorWithDomain:PVAzaharErrorDomain code:2 userInfo:@{NSLocalizedDescriptionKey: @"Azahar rejected this save state (different core version or title)"}];
        return NO;
    }
    return YES;
}

- (void)saveStateToFileAtPath:(NSString *)fileName completionHandler:(SaveStateCompletion)block {
    NSString *path = [fileName copy];
    dispatch_async(dispatch_get_global_queue(QOS_CLASS_USER_INITIATED, 0), ^{
        NSError *e = nil; const BOOL ok = [self saveStateToFileAtPath:path error:&e]; block(ok ? nil : e);
    });
}
- (void)loadStateFromFileAtPath:(NSString *)fileName completionHandler:(SaveStateCompletion)block {
    NSString *path = [fileName copy];
    dispatch_async(dispatch_get_global_queue(QOS_CLASS_USER_INITIATED, 0), ^{
        NSError *e = nil; const BOOL ok = [self loadStateFromFileAtPath:path error:&e]; block(ok ? nil : e);
    });
}
@end
```

```swift
import Foundation
import PVLogging

/// Moves emuThreeDS's 3DS user data (nand, sdmc, sysdata, config, cheats, shaders) from the
/// legacy root into the Azahar root. Same-volume moves are renames, so no disk is duplicated.
/// `states` is skipped on purpose: emuThreeDS save states cannot be loaded by azahar.
public struct PVAzaharDataMigrator {
    public enum Action: Equatable { case move, skipMissing, skipExists }
    public struct Plan {
        public let items: [(directory: String, action: Action)]
        public let totalBytes: Int64
        public var hasWork: Bool { items.contains { $0.action == .move } }
    }
    public enum Error: Swift.Error { case moveFailed(directory: String, underlying: Swift.Error) }

    public static let candidates = ["nand", "sdmc", "sysdata", "config", "cheats", "shaders"]
    public static let markerName = ".migrated-from-emuthree"
    private static let emuThreeCoreClassName = "PVEmuThree.PVEmuThreeCore"

    public let legacyRoot: URL
    public let targetRoot: URL
    private let fm: FileManager

    public init(legacyRoot: URL, targetRoot: URL, fileManager: FileManager = .default) {
        self.legacyRoot = legacyRoot; self.targetRoot = targetRoot; self.fm = fileManager
    }

    public static func defaultLegacyRoot() -> URL {
        #if os(tvOS)
        return FileManager.default.urls(for: .cachesDirectory, in: .userDomainMask)[0]
        #else
        return FileManager.default.urls(for: .documentDirectory, in: .userDomainMask)[0]
        #endif
    }
    public static func defaultTargetRoot() -> URL {
        FileManager.default.urls(for: .documentDirectory, in: .userDomainMask)[0].appendingPathComponent("Azahar")
    }
    public static var emuThreeCoreIsPresent: Bool { NSClassFromString(emuThreeCoreClassName) != nil }

    public var alreadyMigrated: Bool { fm.fileExists(atPath: targetRoot.appendingPathComponent(Self.markerName).path) }

    public func plan() -> Plan {
        if alreadyMigrated { return Plan(items: [], totalBytes: 0) }
        var items: [(String, Action)] = []
        var bytes: Int64 = 0
        for dir in Self.candidates {
            let src = legacyRoot.appendingPathComponent(dir), dst = targetRoot.appendingPathComponent(dir)
            var isDir: ObjCBool = false
            guard fm.fileExists(atPath: src.path, isDirectory: &isDir), isDir.boolValue else { items.append((dir, .skipMissing)); continue }
            if fm.fileExists(atPath: dst.path) { items.append((dir, .skipExists)); continue }
            items.append((dir, .move)); bytes += directorySize(src)
        }
        return Plan(items: items, totalBytes: bytes)
    }

    @discardableResult
    public func apply() throws -> Plan {
        let plan = plan()
        guard plan.hasWork else { return plan }
        try fm.createDirectory(at: targetRoot, withIntermediateDirectories: true)
        for (dir, action) in plan.items where action == .move {
            let src = legacyRoot.appendingPathComponent(dir), dst = targetRoot.appendingPathComponent(dir)
            do { try fm.moveItem(at: src, to: dst) }
            catch {
                ILOG("[PVAzahar] rename of \(dir) failed (\(error)); copying instead")
                do { try fm.copyItem(at: src, to: dst); try fm.removeItem(at: src) }
                catch { throw Error.moveFailed(directory: dir, underlying: error) }
            }
        }
        fm.createFile(atPath: targetRoot.appendingPathComponent(Self.markerName).path, contents: Date().description.data(using: .utf8))
        return plan
    }

    private func directorySize(_ url: URL) -> Int64 {
        guard let e = fm.enumerator(at: url, includingPropertiesForKeys: [.fileSizeKey, .isRegularFileKey]) else { return 0 }
        var total: Int64 = 0
        for case let f as URL in e {
            if let v = try? f.resourceValues(forKeys: [.fileSizeKey, .isRegularFileKey]), v.isRegularFile == true { total += Int64(v.fileSize ?? 0) }
        }
        return total
    }
}
```
