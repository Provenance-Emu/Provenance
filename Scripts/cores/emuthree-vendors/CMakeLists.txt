cmake_minimum_required(VERSION 3.16)
project(PVEmuThreeVendors LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_POLICY_DEFAULT_CMP0077 NEW)
set(CMAKE_POLICY_DEFAULT_CMP0167 OLD)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ARCHITECTURE arm64)
set(Boost_NO_SYSTEM_PATHS ON)
set(BOOST_ROOT "${PV_SOURCES}/boost")
set(Boost_INCLUDE_DIR "${PV_SOURCES}/boost" CACHE PATH "" FORCE)
add_library(boost INTERFACE)
target_include_directories(boost SYSTEM INTERFACE "${PV_SOURCES}/boost")
foreach(component iostreams program_options serialization)
    if(component STREQUAL "iostreams")
        set(sources)
        foreach(unit file_descriptor mapped_file bzip2 gzip zlib)
            list(APPEND sources "${PV_SOURCES}/boost/libs/iostreams/src/${unit}.cpp")
        endforeach()
    else()
        file(GLOB sources "${PV_SOURCES}/boost/libs/${component}/src/*.cpp")
    endif()
    add_library(boost_${component} STATIC ${sources})
    target_link_libraries(boost_${component} PUBLIC boost)
endforeach()
set(FMT_INSTALL ON CACHE BOOL "" FORCE)
add_subdirectory("${PV_SOURCES}/fmt" fmt EXCLUDE_FROM_ALL)
set(DYNARMIC_TESTS OFF CACHE BOOL "" FORCE)
set(DYNARMIC_FRONTENDS A32 CACHE STRING "" FORCE)
set(DYNARMIC_WARNINGS_AS_ERRORS OFF CACHE BOOL "" FORCE)
add_subdirectory("${PV_SOURCES}/dynarmic" dynarmic EXCLUDE_FROM_ALL)
# Reuse the existing mobile memory-protection adapter, leaving upstream Oaknut untouched.
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/mobile/oaknut")
configure_file("${PV_REPO}/Cores/emuThree/lib/oaknut/code_block.hpp"
    "${CMAKE_CURRENT_BINARY_DIR}/mobile/oaknut/code_block.hpp" COPYONLY)
target_include_directories(dynarmic BEFORE PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/mobile")
set(SKIP_GLSLANG_INSTALL ON CACHE BOOL "" FORCE)
set(ENABLE_GLSLANG_BINARIES OFF CACHE BOOL "" FORCE)
set(ENABLE_SPVREMAPPER OFF CACHE BOOL "" FORCE)
set(ENABLE_CTEST OFF CACHE BOOL "" FORCE)
add_subdirectory("${PV_SOURCES}/glslang" glslang EXCLUDE_FROM_ALL)
add_subdirectory("${PV_SOURCES}/sirit" sirit EXCLUDE_FROM_ALL)
set(ENET_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${PV_SOURCES}/enet" enet EXCLUDE_FROM_ALL)
set(TEAKRA_BUILD_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(TEAKRA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
add_subdirectory("${PV_SOURCES}/teakra" teakra EXCLUDE_FROM_ALL)
set(ZSTD_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZSTD_LEGACY_SUPPORT OFF CACHE BOOL "" FORCE)
add_subdirectory("${PV_REPO}/Cores/emuThree/cmake/zstd/build/cmake" zstd EXCLUDE_FROM_ALL)
file(GLOB soundtouch_sources "${PV_SOURCES}/soundtouch/source/SoundTouch/*.cpp")
add_library(SoundTouch STATIC ${soundtouch_sources})
configure_file("${CMAKE_CURRENT_SOURCE_DIR}/SoundTouchConfig.h"
    "${CMAKE_CURRENT_BINARY_DIR}/soundtouch_config.h" COPYONLY)
target_include_directories(SoundTouch PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" PUBLIC "${PV_SOURCES}/soundtouch/include")

set(LIBTYPE STATIC CACHE STRING "" FORCE)
set(ALSOFT_UTILS OFF CACHE BOOL "" FORCE)
set(ALSOFT_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ALSOFT_TESTS OFF CACHE BOOL "" FORCE)
set(ALSOFT_INSTALL OFF CACHE BOOL "" FORCE)
set(ALSOFT_CONFIG OFF CACHE BOOL "" FORCE)
add_subdirectory("${PV_SOURCES}/openal" openal EXCLUDE_FROM_ALL)
